\begin{frame}{Prose colons}
\textbf{Note}:
A lone colon is no suite marker, so the line above is raw TeX.
\emph{Caption}{%
A real marker turns the same shape into a structured command.
}
\begin{itemize}
\item Summary:
    An indented block under a prose colon is raw TeX too, so a
    list item reads the way TeX reads it.
\item \textbf{Detail}:
  A shallower continuation line leaves the colon as text.
\end{itemize}
\cmd{%
\textbf{Term}:
}{%
\textbf{Body}:
continues on a continuation line
}
\end{frame}
